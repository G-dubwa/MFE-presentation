\begin{frame}[t]{Why the frozen-volatility proxy drifts}
  \begin{beamercolorbox}[rounded=true,sep=0.10cm]{block body}
    \centering
    {\small\textbf{Frozen-volatility liquid hedge-option proxy:}\quad
    \(g(t,S,v)=C_{\bs}\!\left(S,K_h,T_h-t,\sqrt v\right)\)}
  \end{beamercolorbox}
  \vspace{0.06cm}
  {\scriptsize\bfseries Apply Itô's lemma under the actual Heston dynamics to the Black--Scholes pricing function \(g(t,S,v)\).}
  \vspace{0.07cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \pill[Teal]{Cancelled by the frozen-\(v\) Black--Scholes PDE}
      \vspace{0.05cm}
      \begin{beamercolorbox}[rounded=true,sep=0.12cm,wd=0.98\linewidth]{block body}
        \centering\(\displaystyle g_t+rSg_S+\frac12vS^2g_{SS}-rg=0\)
      \end{beamercolorbox}
      \vspace{0.07cm}
      {\scriptsize Repricing the level at every date does not make the resulting process satisfy the Heston pricing equation.}
    \end{column}
    \begin{column}{0.50\textwidth}
      \pill[Coral]{Residual discounted drift under Heston}
      \vspace{0.05cm}
      \begin{beamercolorbox}[rounded=true,sep=0.12cm,wd=0.98\linewidth]{block body}
        \centering{\fontsize{8.6}{10}\selectfont\(\displaystyle
          R(t,S,v)=\kappa(\theta-v)g_v
          +\frac12\xi^2vg_{vv}
          +\rho\xi vSg_{Sv}\)}
      \end{beamercolorbox}
      \vspace{0.07cm}
      {\scriptsize At submitted \(r=0\), discounted and undiscounted residuals coincide.}
    \end{column}
  \end{columns}
  \vspace{0.09cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.32\textwidth}
      \centering\pill[Teal]{1}\quad{\scriptsize mean reversion \(\times g_v\)}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering\pill[Gold]{2}\quad{\scriptsize vol-of-vol \(\times g_{vv}\)}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering\pill[Coral]{3}\quad{\scriptsize spot--variance cross term}
    \end{column}
  \end{columns}
  \vspace{0.07cm}
  \begin{beamercolorbox}[rounded=true,sep=0.11cm]{block body}
    {\scriptsize\textbf{Scope:} The sign is state-dependent. Under the submitted parameters and simulated states, the average residual was negative, consistent with the martingale diagnostic.}
  \end{beamercolorbox}
  \sourcefoot{Submitted report, Sec.\ 4.1.4 and App.\ A.6.3; Black--Scholes (1973); Heston (1993).}
  \note{Define the proxy as the liquid hedging option, apply Itô under Heston, then explain the frozen-v PDE cancellation and the three remaining drift terms.}
\end{frame}

\begin{frame}[t]{COS evaluates a Heston-consistent option price}
  \begin{beamercolorbox}[rounded=true,sep=0.09cm]{block body}
    \centering
    {\fontsize{8.7}{10}\selectfont\(\displaystyle
      C^H(t,S,v)=\mathbb E^{\mathbb Q}\!\left[
        e^{-r(T_h-t)}(S_{T_h}-K_h)^+

      \right]\)}
  \end{beamercolorbox}

  {\fontsize{7.2}{8.3}\selectfont\textbf{Bridge:} Heston supplies a closed-form characteristic function but no convenient transition density. COS converts transform values into a deterministic finite cosine sum---there is no nested future-path simulation.\par}

      \vspace{0.08cm}
      {\fontsize{8.2}{9.8}\selectfont
      \begin{enumerate}
        \setlength{\itemsep}{0.06cm}
        \item \textbf{Black--Scholes:} the neural strategy recovered the known finite-grid hedge.
        \item \textbf{Conditioning:} adding the missing contract information repaired the identified generalisation failure.

      \end{enumerate}}
    \end{column}
    \begin{column}{0.41\textwidth}
      \pill[Coral]{What they do not establish}
      \vspace{0.08cm}
      {\fontsize{8.2}{9.8}\selectfont
      \begin{tightitemize}
        \item high-powered statistical evidence beyond the three submitted seeds;
        \item fully out-of-sample premium estimation or superiority to a finite-grid multi-instrument optimum;
        \item performance beyond the frictionless simulated Heston setting and submitted information assumptions.
      \end{tightitemize}}
    \end{column}
  \end{columns}
  \vspace{0.04cm}
  \begin{beamercolorbox}[rounded=true,sep=0.08cm]{block body}
