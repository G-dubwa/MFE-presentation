\begin{frame}[t]{COS evaluates a Heston-consistent option price}
  \begin{beamercolorbox}[rounded=true,sep=0.10cm]{block body}
    \centering
    {\fontsize{8.8}{10.2}\selectfont\(\displaystyle
      C^H(t,S,v)=\mathbb E^{\mathbb Q}\!\left[
        e^{-r(T_h-t)}(S_{T_h}-K_h)^+
        \mid S_t=S,\ v_t=v
      \right]\)}
  \end{beamercolorbox}
  \vspace{0.08cm}
  {\fontsize{7.5}{8.8}\selectfont
    \textbf{Heston supplies the characteristic function; COS maps each simulated Heston state to a liquid-option price and the sensitivities needed for hedging.}\par}
  \vspace{0.11cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \pill{Pricing and marking}
      \vspace{0.06cm}
      \centering{\fontsize{8.4}{9.7}\selectfont\bfseries
        \(\text{Heston characteristic function}
        \longrightarrow \text{COS price and sensitivities}\)\par}
      \vspace{0.13cm}
      \begin{beamercolorbox}[rounded=true,sep=0.10cm]{block body}
        \centering{\fontsize{7.4}{8.6}\selectfont
          \(\text{Monte Carlo Heston states}
          \longrightarrow \text{COS option marks}
          \longrightarrow \text{hedge positions}\)}
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.49\textwidth}
      \pill[Gold]{Exposure-matching positions}
      \vspace{0.06cm}
      \begin{beamercolorbox}[rounded=true,sep=0.11cm]{block body}
        \centering
        {\small\(\displaystyle
          \boxed{\eta_n=\frac{V_n^T}{V_n^h},\qquad
          \delta_n=\Delta_n^T-\eta_n\Delta_n^h}\)}
      \end{beamercolorbox}
      \vspace{0.05cm}
      {\fontsize{7.2}{8.4}\selectfont
        \(V=\partial C/\partial v\). The option holding matches variance exposure; the stock holding removes the remaining stock exposure.\par}
    \end{column}
  \end{columns}
  \vspace{0.13cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \pill{Heston COS benchmark}\quad
      {\fontsize{7.1}{8.3}\selectfont positions from Heston-consistent sensitivities.}
    \end{column}
    \begin{column}{0.49\textwidth}
      \pill[Gold]{BS-proxy Greek heuristic}\quad
      {\fontsize{7.1}{8.3}\selectfont positions from frozen-volatility Black--Scholes sensitivities.}
    \end{column}
  \end{columns}
  \vspace{0.13cm}
  \centering{\small\color{Teal}\bfseries Same traded COS option path; different position rule.\par}
  \sourcefoot{Submitted report, Sec.\ 4.1.4 and Apps.\ A.6.1--A.6.2; Heston (1993); Fang \& Oosterlee (2008).}

\begin{frame}[t]{Three conclusions survive the diagnostics}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \pill{What the experiments establish}
      \vspace{0.10cm}
      {\fontsize{8.2}{9.8}\selectfont
      \begin{enumerate}
        \setlength{\itemsep}{0.10cm}
        \item \textbf{Black--Scholes:} the neural strategy recovered the known finite-grid hedge.
        \item \textbf{Conditioning:} adding the missing contract information repaired the identified generalisation failure.
        \item \textbf{Heston:} with a consistently priced liquid option, the stock-plus-option NN achieved lower terminal error than the strongest tested analytic Greek heuristic in every submitted seed.
      \end{enumerate}}
    \end{column}
    \begin{column}{0.41\textwidth}
      \pill[Coral]{What they do not establish}
      \vspace{0.10cm}
      {\fontsize{8.2}{9.8}\selectfont
      \begin{tightitemize}
        \item high-powered statistical evidence beyond the three submitted seeds;
        \item a derived finite-grid optimal stock-and-option hedge;
        \item performance beyond the frictionless simulated Heston setting and submitted information assumptions.
      \end{tightitemize}}
    \end{column}
  \end{columns}
