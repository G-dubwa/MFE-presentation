\begin{frame}{Backup: the discrete-time conditional projection}
  \begin{align*}
    \min_{\pi,\delta}\quad&
    \mathbb{E}\!\left[
      \left(\pi+\sum_{j=0}^{N-1}\delta_j\Delta S_j-H\right)^2
    \right],\\
    0={}&
    \mathbb{E}\!\left[
      \left(H-\pi-\sum_{j=0}^{N-1}\delta_j\Delta S_j\right)
      \Delta S_n\mid\mathcal{F}_{t_n}
    \right].
  \end{align*}
  Under the submitted \(r=0\) risk-neutral simulation, \(S\) is a martingale, so
  \[
    \deltadt_n =
    \frac{\mathbb{E}[H\Delta S_n\mid\mathcal{F}_{t_n}]}
         {\mathbb{E}[(\Delta S_n)^2\mid\mathcal{F}_{t_n}]}
    =
    \frac{\operatorname{Cov}(H,\Delta S_n\mid S_{t_n})}
         {\operatorname{Var}(\Delta S_n\mid S_{t_n})}.
  \]
  \risknote{This simplification is tied to the submitted \(r=0\) martingale setting; local and global quadratic hedges need not coincide more generally.}
  \vspace{-0.12cm}
  \sourcefoot{Submitted report, Appendix A.4; Schweizer (1995, 2001).}
\end{frame}

\begin{frame}[t]{\texorpdfstring{\raisebox{-0.25cm}[0pt][0pt]{Frozen-volatility drift derivation}}{Frozen-volatility drift derivation}}
  {\scriptsize For \(g(t,S,v)=C_{\bs}(S,K_h,T_h-t,\sqrt v)\), the Heston multiplication rules are}
  {\small\[
    (dS)^2=vS^2dt,
    \qquad (dv)^2=\xi^2vdt,
    \qquad dS\,dv=\rho\xi vSdt.
  \]}
  {\fontsize{8.5}{10}\selectfont\[
  \begin{aligned}
    dg={}&\Big[g_t+rSg_S+\kappa(\theta-v)g_v+\tfrac12vS^2g_{SS}
      +\tfrac12\xi^2vg_{vv}+\rho\xi vSg_{Sv}\Big]dt\\
      &+g_SS\sqrt v\,dW^S+g_v\xi\sqrt v\,dW^v .
  \end{aligned}
  \]}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \pill[Teal]{Frozen-\(v\) BS PDE cancellation}
      \vspace{0.05cm}
      \centering{\small\(\displaystyle g_t+rSg_S+\tfrac12vS^2g_{SS}-rg=0\)}
    \end{column}
    \begin{column}{0.50\textwidth}
      \pill[Coral]{Remaining Heston residual}
      \vspace{0.05cm}
      \centering{\small\(\displaystyle \kappa(\theta-v)g_v+\tfrac12\xi^2vg_{vv}+\rho\xi vSg_{Sv}\)}
    \end{column}
  \end{columns}
  \begin{beamercolorbox}[rounded=true,sep=0.12cm]{block body}
    {\scriptsize With submitted \(r=0\), this is both the discounted and undiscounted drift residual. Its sign is state-dependent; the diagnostic is not a complete formal arbitrage proof.}
  \end{beamercolorbox}
  \sourcefoot{Submitted report, Appendix A.6.3.}
\end{frame}

\begin{frame}[t]{COS pricing}

        C_{\text{COS}}
        &\approx e^{-r\tau}
        \sum_{k=0}^{N_{\text{COS}}-1}{}' A_kV_k,\\
        A_k&=\frac{2}{b-a}\Re\!\left[

\begin{frame}[fragile]{Heston simulation}
\begin{lstlisting}[style=deckcode]
for n in range(N_STEPS):
    z1 = torch.randn(n_paths, device=device)
    z2 = torch.randn(n_paths, device=device)
    z_s = z1
    z_v = RHO*z1 + math.sqrt(max(1.0-RHO**2, 0.0))*z2
    v_pos = torch.clamp(v[:, n], min=0.0)
    S[:, n+1] = S[:, n] * torch.exp(
        (R - 0.5*v_pos)*dt + torch.sqrt(v_pos)*sqrt_dt*z_s
    )
    v_next = v[:, n] + KAPPA*(THETA-v_pos)*dt \
             + XI*torch.sqrt(v_pos)*sqrt_dt*z_v
    v[:, n+1] = torch.clamp(v_next, min=0.0)
\end{lstlisting}
  \vspace{-0.08cm}
  {\scriptsize
  \begin{tabular}{@{}p{0.18\linewidth}p{0.76\linewidth}@{}}
    \texttt{z\_s} / \texttt{z\_v} & stock shock / correlated variance shock\\
    \texttt{v\_pos} & non-negative variance used in the drift, diffusion and stock log-Euler update\\
    final clamp & prevents a negative Euler proposal becoming the next stored variance\\
  \end{tabular}}
  \vspace{0.06cm}
  \risknote{The Feller condition \(2\kappa\theta=0.64\geq\xi^2=0.36\) concerns the continuous-time process; it does not guarantee that an Euler proposal is never negative. Train, validation and test paths are simulated independently.}
  \sourcefoot{Adapted only for line length from submitted report Listing C.4 and Algorithm A.3.3. Primary speaker: Siphelele.}
\end{frame}

\begin{frame}[fragile]{Two-instrument gain construction}
\begin{lstlisting}[style=deckcode]
# state at t_n only
x_n = [log(S_n/K), (T-t_n)/T, v_n/THETA,
       log(S_n/K_H), (T_H-t_n)/T_H, C_h_n/S_n]
z_stock, z_option = shared_mlp(x_n)
delta_n = 5.0 * tanh(z_stock)
eta_n   = 5.0 * tanh(z_option)

gain = sum(delta_n * (S_next-S_n)
           + eta_n * (C_h_next-C_h_n), over="time")
hedge_error = premium + gain - maximum(S_T-K, 0.0)
loss = mean(hedge_error**2)
