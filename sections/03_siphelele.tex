\begin{frame}{Heston introduces a second source of risk}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=0.18cm]{block body}
        \[
          \begin{aligned}
          dS_t &= rS_t\,dt+\sqrt{v_t}S_t\,dW_t^S,\\
          dv_t &= \kappa(\theta-v_t)\,dt+\xi\sqrt{v_t}\,dW_t^v,\\
          dW_t^S\,dW_t^v&=\rho\,dt.
          \end{aligned}
        \]
      \end{beamercolorbox}
      \vspace{0.12cm}
      \[
        dW_t^v=\rho\,dW_t^S+\sqrt{1-\rho^2}\,dW_t^\perp
      \]
      \vspace{-0.04cm}
      {\small\begin{tightitemize}
        \item \(v_t\) is instantaneous variance; \(\sqrt{v_t}\) is volatility.
        \item Since \(|\rho|<1\), an orthogonal variance shock remains.
        \item Stock supplies one traded diffusion exposure; a variance-sensitive liquid option supplies a second locally independent exposure.
      \end{tightitemize}}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[
        src/.style={circle,minimum size=0.82cm,fill=Coral!12,draw=Coral!30,font=\small\bfseries},
        asset/.style={rounded corners=4pt,minimum width=2.45cm,minimum height=0.72cm,
          fill=Mist,draw=DeepNavy!20,font=\small\bfseries},
        arr/.style={-{Latex[length=2mm]},line width=0.9pt,color=Teal}
      ]
        \node[src] (ws) at (0,1.65) {\(W^S\)};
        \node[src] (wv) at (2.55,1.65) {\(W^v\)};
        \draw[dashed,line width=0.8pt,color=Coral!80] (ws)--node[above,font=\scriptsize,color=Slate] {\(\rho=-0.70\)} (wv);
        \node[asset] (stock) at (0,0.15) {stock};
        \node[asset,fill=Teal!12] (option) at (1.27,-1.25) {liquid option};
        \draw[arr] (ws)--(stock);
        \draw[arr] (ws)--(option);
        \draw[arr] (wv)--(option);
      \end{tikzpicture}
      \vspace{0.18cm}
      \begin{beamercolorbox}[rounded=true,sep=0.13cm]{block body}
        {\scriptsize\textbf{Scope:} Local spanning holds in the ideal continuous-time frictionless model; discrete trading still leaves hedging error.}
      \end{beamercolorbox}
    \end{column}
  \end{columns}
  \sourcefoot{Submitted report, Secs.\ 2.1.2 and 4.1.4; Heston (1993). Feller calculation in backup.}
  \note{Explain the orthogonal variance component and qualify the local spanning statement.}
\end{frame}

\begin{frame}{How Heston training paths are generated}
  \centering
  \begin{tikzpicture}[
    box/.style={rounded corners=3pt,draw=DeepNavy!18,fill=Mist,
      minimum height=0.64cm,text width=1.62cm,align=center,font=\scriptsize},
    arr/.style={-{Latex[length=2mm]},color=Teal,line width=0.8pt},
    node distance=0.14cm
  ]
    \node[box] (z) {independent\\\((Z_n^S,Z_n^\perp)\)};
    \node[box,right=of z] (zv) {correlate\\\(Z_n^v\)};
    \node[box,right=of zv] (sv) {update\\\((S_{t_n},v_{t_n})\)};
    \node[box,right=of sv] (c) {reprice\\\(C_{t_n}^h\)};
    \node[box,right=of c] (x) {form state\\\(x_n\)};
    \node[box,right=of x,fill=Teal!12] (pos) {choose\\\((\delta_n,\eta_n)\)};
    \draw[arr] (z)--(zv);\draw[arr] (zv)--(sv);\draw[arr] (sv)--(c);
    \draw[arr] (c)--(x);\draw[arr] (x)--(pos);
  \end{tikzpicture}
  \vspace{0.13cm}
  \[
    Z_n^v=\rho Z_n^S+\sqrt{1-\rho^2}\,Z_n^\perp,
    \qquad
    (\delta,\eta)\longrightarrow G_T\longrightarrow HE.
  \]
  \vspace{-0.04cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \pill{Path construction}
      {\small\begin{tightitemize}
        \item log-Euler updates stock;
        \item full-truncation Euler updates variance, so negative Euler values do not enter the square root or next update;
        \item the liquid option is repriced at every hedge date.
      \end{tightitemize}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \pill[Gold]{Training sample}
      {\small\begin{tightitemize}
        \item train, validation and test paths are simulated independently;
        \item one sample is an entire trajectory, not one isolated observation;
        \item \(S,v,C^h\!:\,B\!\times\!(N+1)\), \(x\!:\,B\!\times\!N\!\times\!6\).
      \end{tightitemize}}
    \end{column}
  \end{columns}
  \sourcefoot{Submitted report, Listing C.4 and Algorithms A.3.3, A.6.8, A.6.10.}
  \note{Explain that every path is simulated rather than historical and becomes one terminal-error observation.}
\end{frame}

\begin{frame}{The hedge chooses two traded positions}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      {\fontsize{9.2}{10.5}\selectfont\[
        G_T=\sum_{n=0}^{N-1}\delta_n\Delta S_n
        +\sum_{n=0}^{N-1}\eta_n\Delta C_n^h
      \]}
      \vspace{-0.04cm}
      \begin{beamercolorbox}[rounded=true,sep=0.12cm,wd=0.98\linewidth]{block body}
        {\small\(\delta_n,\eta_n\) are desired \textbf{total holdings} over \([t_n,t_{n+1}]\), not trades.}
      \end{beamercolorbox}
      \vspace{0.12cm}
      {\scriptsize
      \begin{tabular}{@{}p{0.27\linewidth}p{0.65\linewidth}@{}}
        \toprule
        Target state & \(\log(S_n/K),\ (T-t_n)/T\)\\
        Variance state & \(v_n/\theta\)\\
        Hedge instrument & \(\log(S_n/K_h),\ (T_h-t_n)/T_h,\ C_n^h/S_n\)\\
        \bottomrule
      \end{tabular}}
    \end{column}
    \begin{column}{0.40\textwidth}
      \pill{Signed, bounded holdings}
      {\fontsize{8.7}{10}\selectfont\[
        \delta_n=5\tanh(z_{1,n}),\quad
        \eta_n=5\tanh(z_{2,n}).
      \]}
      {\scriptsize\begin{tightitemize}
        \item Both holdings may be long or short, so \([0,1]\) is inappropriate.
        \item \([-5,5]\) stabilises training; it is not an optimality theorem.
      \end{tightitemize}}
      \vspace{0.08cm}
      \pill[Gold]{Two different contracts}
      {\scriptsize
      \begin{align*}
        \text{target: }&K=0.9,\ T=0.5,\\
        \text{hedge: }&K_h=1,\ T_h=1.
      \end{align*}
      Near ATM and longer dated: variance-sensitive and alive throughout.}
    \end{column}
  \end{columns}
  \risknote{Full-information simulation: true \(v_n\) enters the state; instantaneous variance is latent in real markets.}
  \sourcefoot{Submitted report, Sec.\ 4.1.4 and Algorithm A.6.10.}
  \note{Distinguish holdings from trade increments and explain why the two contracts differ.}
\end{frame}

\begin{frame}{Frozen-volatility repricing is inconsistent}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=0.22cm]{block body}
        \[
          C_t^h\approx
          C_{\bs}\!\left(S_t,K_h,T_h-t,\sqrt{v_t}\right)
        \]
      \end{beamercolorbox}
      \vspace{0.16cm}
      \begin{tikzpicture}[
        box/.style={rounded corners=3pt,draw=DeepNavy!18,fill=Mist,
          minimum height=0.72cm,text width=2.0cm,align=center,font=\scriptsize},
        arr/.style={-{Latex[length=2mm]},color=Teal,line width=0.8pt},node distance=0.25cm]
        \node[box] (now) {at \(t_n\): insert\\\(\sigma=\sqrt{v_{t_n}}\)};
        \node[box,right=of now] (freeze) {price as if \(\sigma\)\\stays fixed to \(T_h\)};
        \node[box,right=of freeze] (next) {at \(t_{n+1}\): repeat\\with new variance};
        \draw[arr] (now)--(freeze);\draw[arr] (freeze)--(next);
      \end{tikzpicture}
      \vspace{0.16cm}
      \claimbox{Recomputing the level at every date does not make the path satisfy the Heston pricing equation.}
    \end{column}
    \begin{column}{0.39\textwidth}
      \pill{Why it seemed reasonable}
      {\small\begin{tightitemize}
        \item cheap to evaluate;
        \item responsive to current variance;
        \item creates a variance-sensitive second instrument.
      \end{tightitemize}}
      \vspace{0.14cm}
      \pill[Coral]{Process mismatch}
      {\small\begin{tightitemize}
        \item stock and variance follow Heston;
        \item the proxy uses frozen-volatility Black--Scholes valuation;
        \item final COS results use a different traded-option process.
      \end{tightitemize}}
    \end{column}
  \end{columns}
  \sourcefoot{Submitted report, Sec.\ 4.1.4 (pp.\ 24--25) and Algorithm A.6.5.}
  \note{The preliminary proxy RMSE 0.008198 is motivation only and is not comparable with the final COS experiment.}
\end{frame}
