\begin{frame}{Heston adds a risk the stock cannot span}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=0.18cm]{block body}
        \[
          \begin{aligned}
          dS_t &= rS_t\,dt+\sqrt{v_t}S_t\,dW_t^S,\\
          dv_t &= \kappa(\theta-v_t)\,dt+\xi\sqrt{v_t}\,dW_t^v,\\
          dW_t^S\,dW_t^v&=\rho\,dt.
          \end{aligned}
        \]
      \end{beamercolorbox}

      \vspace{0.10cm}
      \[
        dW_t^v=\rho\,dW_t^S+\sqrt{1-\rho^2}\,dW_t^\perp
      \]

      \vspace{-0.06cm}
      {\small\begin{tightitemize}
        \item \(v_t\) is instantaneous variance; \(\sqrt{v_t}\) is volatility.
        \item Since \(|\rho|<1\), an orthogonal variance shock remains.
        \item With \(\rho=-0.70\), the stock cannot span the full variance shock.
      \end{tightitemize}}
    \end{column}

    \begin{column}{0.38\textwidth}
      \pill{Diffusion exposures}
      {\scriptsize
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{@{}lcc@{}}
        \toprule
        Instrument & \(dW^S\) & \(dW^\perp\)\\
        \midrule
        stock & yes & no\\
        liquid option & yes & yes\\
        \bottomrule
      \end{tabular}}

      \vspace{0.14cm}
      \begin{beamercolorbox}[rounded=true,sep=0.14cm]{block body}
        {\scriptsize A liquid option with
        \[
          \frac{\partial C^h}{\partial v}\neq 0
        \]
        supplies a second locally independent exposure.}
      \end{beamercolorbox}

      \vspace{0.14cm}
      \begin{beamercolorbox}[rounded=true,sep=0.13cm]{block body}
        {\scriptsize\textbf{Scope:} Local spanning holds in the ideal continuous-time frictionless model; discrete trading still leaves hedging error.}
      \end{beamercolorbox}
    \end{column}
  \end{columns}

  \sourcefoot{Submitted report, Secs.\ 2.1.2 and 4.1.4; Heston (1993). Feller calculation in backup.}
  \note{Open by saying that Heston changes the problem from one Brownian shock to two. Emphasise that the stock spans the stock shock, but not the orthogonal variance shock. Conclude that a second variance-sensitive traded instrument is needed.}
\end{frame}

\begin{frame}{Each training sample is a full Heston trajectory}
  \centering
  \begin{tikzpicture}[
    box/.style={rounded corners=3pt,draw=DeepNavy!18,fill=Mist,
      minimum height=0.60cm,text width=1.55cm,align=center,font=\scriptsize},
    arr/.style={-{Latex[length=2mm]},color=Teal,line width=0.8pt},
    node distance=0.13cm
  ]
    \node[box] (z) {independent\\\((Z_n^S,Z_n^\perp)\)};
    \node[box,right=of z] (zv) {correlate\\\(Z_n^v\)};
    \node[box,right=of zv] (sv) {update\\\((S_n,v_n)\)};
    \node[box,right=of sv] (c) {reprice\\\(C_n^h\)};
    \node[box,right=of c] (x) {form state\\\(x_n\)};
    \node[box,right=of x,fill=Teal!12] (pos) {choose\\\((\delta_n,\eta_n)\)};
    \draw[arr] (z)--(zv);
    \draw[arr] (zv)--(sv);
    \draw[arr] (sv)--(c);
    \draw[arr] (c)--(x);
    \draw[arr] (x)--(pos);
  \end{tikzpicture}

  \vspace{0.08cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.51\textwidth}
      \pill{Simulation step}
      {\fontsize{7.5}{8.8}\selectfont
      \[
        Z_n^v=\rho Z_n^S+\sqrt{1-\rho^2}\,Z_n^\perp .
      \]
      \[
        S_{n+1}=S_n
        \exp\!\left[
        \left(r-\frac12v_n^+\right)\Delta t
        +\sqrt{v_n^+\Delta t}\,Z_n^S
        \right].
      \]
      \[
        \widetilde v_{n+1}
        =v_n+\kappa(\theta-v_n^+)\Delta t
        +\xi\sqrt{v_n^+\Delta t}\,Z_n^v,
      \]
      \[
        v_{n+1}=\max(\widetilde v_{n+1},0).
      \]}
    \end{column}

    \begin{column}{0.45\textwidth}
      \pill[Gold]{Network state}
      {\fontsize{7.5}{8.8}\selectfont
      \[
        x_n=
        \left(
        \log\frac{S_n}{K},
        \frac{T-t_n}{T},
        \frac{v_n}{\theta},
        \log\frac{S_n}{K_h},
        \frac{T_h-t_n}{T_h},
        \frac{C_n^h}{S_n}
        \right).
      \]}

      \vspace{0.08cm}
      {\small\begin{tightitemize}
        \item Log-Euler keeps \(S_{n+1}>0\).
        \item Full truncation stops negative variance from entering the next square root.
        \item One sample is a complete trajectory, not one isolated observation.
      \end{tightitemize}}
    \end{column}
  \end{columns}

  \risknote{Full-information simulation: true \(v_n\) enters the state; instantaneous variance is latent in real markets.}
  \sourcefoot{Submitted report, Listing C.4 and Algorithms A.3.3, A.6.8, A.6.10.}
  \note{Explain that the training data are simulated full paths. The network acts at each path-date state, and the full path gives one terminal gain and one terminal hedge error.}
\end{frame}

  \begin{columns}[T,onlytextwidth]

          C_t^h\approx

        \]
      \end{beamercolorbox}

      \vspace{0.14cm}
      \begin{tikzpicture}[

      {\small\begin{tightitemize}
        \item cheap to evaluate;
        \item responsive to current variance;
        \item creates a variance-sensitive second instrument.
      \end{tightitemize}}

      \vspace{0.14cm}
      \pill[Coral]{Question}
      {\small
      Does repeatedly recomputing the Black--Scholes level produce a risk-neutral martingale under the simulated Heston dynamics?}
