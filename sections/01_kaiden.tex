\begin{frame}{The hedge is learned through terminal portfolio error}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \vspace{0.15cm}
      \begin{beamercolorbox}[rounded=true,sep=0.25cm]{block body}
        {\fontsize{9}{11}\selectfont
        \[
          \begin{aligned}
          \he
          &=
          \pi+
          \underbrace{\sum_{n=0}^{N-1}\delta_n
          \bigl(S_{t_{n+1}}-S_{t_n}\bigr)}_{\text{self-financing gains}}
          -(S_T-K)^+,\\[-0.05cm]
          &\hspace{1.2cm}\min_{\pi,\theta}\; \mathbb{E}[\he^2].
          \end{aligned}
        \]}
      \end{beamercolorbox}
      \vspace{0.28cm}
      \claimbox{The analytic delta is never used as a training target.}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{tightitemize}
        \item \(N=125\) discrete hedge dates in the baseline.
        \item \(\delta_n=f_\theta(x_n)\) is adapted to information at \(t_n\).
        \item Negative \(\he\) is a seller shortfall; reported seller loss is \(L=-\he\).
        \item With \(r=0\), the submitted experiments use undiscounted gains.
      \end{tightitemize}
      \vspace{0.15cm}
      \begin{tikzpicture}[x=0.75cm,y=0.6cm]
        \draw[-{Latex[length=2mm]},line width=0.8pt,color=Slate] (0,0)--(5.2,0);

          \node[below,font=\scriptsize,color=Slate] at (\x,-0.13) {\(\lab\)};
        }
        \node[above,font=\scriptsize\bfseries,color=DeepNavy] at (2.5,0.22) {rebalance using current state};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \sourcefoot{Submitted report, Secs.\ 1.1, 2.2.2 and 3.1; Eqns.\ (1.1)--(1.2).}
